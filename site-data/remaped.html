Product Data Schema (TypeScript)

// data/types.ts
export interface Product {
  slug: string;
  name: string;
  category: 'vegetable-powders' | 'fruit-powders';
  shortDescription: string;
  longDescription: string;
  image: string;
  specs: {
    color: string;
    aroma: string;
    moisture: string;        // e.g. "Max 5%"
    meshSize: string;        // e.g. "60-80 mesh"
    bulkDensity: string;     // e.g. "0.5-0.7 g/ml"
    shelfLife: string;       // e.g. "24 months"
    packaging: string[];     // e.g. ["25kg bags", "50kg drums"]
    hsCode: string;
    countryOfOrigin: string;
  };
  applications: string[];      // ["Soups", "Seasonings", ...]
  certifications: string[];
  relatedSlugs: string[];
  seo: {
    title: string;
    description: string;
    keywords: string[];
  };
}

SEO — Metadata & generateMetadata

// app/products/vegetable-powders/[slug]/page.tsx
import { Metadata } from 'next';
import { getProductBySlug, getAllVegSlugs } from '@/lib/products';

// Generate all static paths at build time
export async function generateStaticParams() {
  const slugs = await getAllVegSlugs();
  return slugs.map(slug => ({ slug }));
}

// Dynamic metadata per product
export async function generateMetadata(
  { params }: { params: { slug: string } }
): Promise<Metadata> {
  const product = await getProductBySlug(params.slug, 'vegetable');
  return {
    title: product.seo.title,
    description: product.seo.description,
    keywords: product.seo.keywords,
    openGraph: {
      title: product.seo.title,
      description: product.seo.description,
      images: [{ url: product.image, alt: product.name }],
      type: 'website',
    },
    alternates: {
      canonical: `https://srigreen.com/products/vegetable-powders/${params.slug}`,
    },
  };
}

JSON-LD Schema Component

// components/seo/JsonLd.tsx
export function ProductJsonLd({ product }: { product: Product }) {
  const schema = {
    "@context": "https://schema.org",
    "@type": "Product",
    "name": product.name,
    "description": product.shortDescription,
    "image": product.image,
    "brand": { "@type": "Brand", "name": "SRI GREEN Industries" },
    "countryOfOrigin": "IN",
    "category": product.category,
    "offers": {
      "@type": "Offer",
      "availability": "https://schema.org/InStock",
      "url": `https://srigreen.com/products/${product.category}/${product.slug}`,
      "seller": { "@type": "Organization", "name": "SRI GREEN Industries" }
    }
  };
  return (
    <script

Sitemap & Robots

// app/sitemap.ts — auto-generates /sitemap.xml
import { MetadataRoute } from 'next';
import { getAllProducts } from '@/lib/products';

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const products = await getAllProducts();
  const productUrls = products.map(p => ({
    url: `https://srigreen.com/products/${p.category}/${p.slug}`,
    lastModified: new Date(),
    changeFrequency: 'monthly' as const,
    priority: 0.8,
  }));

  return [
    { url: 'https://srigreen.com', priority: 1.0, changeFrequency: 'weekly' },
    { url: 'https://srigreen.com/products', priority: 0.9 },
    { url: 'https://srigreen.com/products/vegetable-powders', priority: 0.9 },
    { url: 'https://srigreen.com/products/fruit-powders', priority: 0.9 },
    { url: 'https://srigreen.com/quality', priority: 0.8 },
    { url: 'https://srigreen.com/about', priority: 0.7 },
    { url: 'https://srigreen.com/contact', priority: 0.9 },
    { url: 'https://srigreen.com/faq', priority: 0.7 },
    ...productUrls,
  ];
}

// app/robots.ts
export default function robots(): MetadataRoute.Robots {
  return {
    rules: { userAgent: '*', allow: '/' },
    sitemap: 'https://srigreen.com/sitemap.xml',
  };
}
